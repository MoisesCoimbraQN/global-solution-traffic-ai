🔍 CONSULTAS SQL ÚTEIS
═════════════════════════════════════════════════════════════════════════════════

1️⃣ Ver Resumo de Tráfego Atual
   SELECT * FROM vw_transito_resumido ORDER BY score_medio DESC;

2️⃣ Eficácia dos Semáforos
   SELECT * FROM vw_eficacia_semaforos ORDER BY eficacia_media DESC;

3️⃣ Horários de Pico
   SELECT EXTRACT(HOUR FROM data_criacao) as hora, 
          AVG(nivel_congestionamento_score) as score
   FROM fato_transito_tempo_real 
   WHERE data_criacao >= NOW() - INTERVAL '7 days'
   GROUP BY hora ORDER BY hora;

4️⃣ Vias Mais Congestionadas
   SELECT v.nome_via, AVG(f.nivel_congestionamento_score) as score
   FROM fato_transito_tempo_real f
   JOIN dim_via v ON f.id_via = v.id_via
   WHERE f.data_criacao >= NOW() - INTERVAL '24 hours'
   GROUP BY v.nome_via
   ORDER BY score DESC;
